### 二、遗物系统管理器

#### 3. 遗物管理器 `RelicManager.cs`

```csharp
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using QFramework;

namespace Gambling
{
    public class RelicManager : MonoBehaviour, IController
    {
        [Header("遗物库配置")]
        [SerializeField] private List<RelicData> allRelics = new List<RelicData>();
        
        // 玩家已拥有的遗物
        private List<RelicData> ownedRelics = new List<RelicData>();
        
        public IArchitecture GetArchitecture()
        {
            return Global.Interface;
        }
        
        /// <summary>
        /// 根据当前关卡抽取遗物
        /// </summary>
        public List<RelicData> DrawRelicsForShop(int currentLevel, int count = 4)
        {
            // 获取已解锁的遗物
            var unlockedRelics = allRelics.Where(r => r.isUnlocked && 
                                                      r.unlockRequiredLevel <= currentLevel).ToList();
            
            if (unlockedRelics.Count == 0) return new List<RelicData>();
            
            // 计算各品质的权重
            Dictionary<RelicQuality, float> qualityWeights = CalculateQualityWeights(currentLevel);
            
            List<RelicData> drawnRelics = new List<RelicData>();
            
            for (int i = 0; i < count; i++)
            {
                RelicQuality selectedQuality = SelectQualityByWeight(qualityWeights);
                RelicData selectedRelic = SelectRelicByQuality(unlockedRelics, selectedQuality);
                
                if (selectedRelic != null)
                {
                    drawnRelics.Add(selectedRelic);
                }
            }
            
            return drawnRelics;
        }
        
        /// <summary>
        /// 计算各品质的权重（关卡越高，高品质概率越大）
        /// </summary>
        private Dictionary<RelicQuality, float> CalculateQualityWeights(int level)
        {
            Dictionary<RelicQuality, float> weights = new Dictionary<RelicQuality, float>();
            
            if (level == 1)
            {
                // 第1关：100%普通品质
                weights[RelicQuality.Common] = 100f;
                weights[RelicQuality.Rare] = 0f;
                weights[RelicQuality.Epic] = 0f;
                weights[RelicQuality.Legendary] = 0f;
            }
            else
            {
                // 每过一关，高品质概率增加
                float levelBonus = (level - 1) * 5f; // 每关增加5%
                
                weights[RelicQuality.Common] = Mathf.Max(60f - levelBonus, 20f);
                weights[RelicQuality.Rare] = Mathf.Min(25f + levelBonus * 0.5f, 40f);
                weights[RelicQuality.Epic] = Mathf.Min(10f + levelBonus * 0.3f, 25f);
                weights[RelicQuality.Legendary] = Mathf.Min(5f + levelBonus * 0.2f, 15f);
            }
            
            return weights;
        }
        
        /// <summary>
        /// 根据权重选择品质
        /// </summary>
        private RelicQuality SelectQualityByWeight(Dictionary<RelicQuality, float> weights)
        {
            float totalWeight = weights.Values.Sum();
            float randomValue = Random.Range(0f, totalWeight);
            float currentWeight = 0f;
            
            foreach (var kvp in weights)
            {
                currentWeight += kvp.Value;
                if (randomValue <= currentWeight)
                {
                    return kvp.Key;
                }
            }
            
            return RelicQuality.Common;
        }
        
        /// <summary>
        /// 从指定品质中随机选择遗物
        /// </summary>
        private RelicData SelectRelicByQuality(List<RelicData> relics, RelicQuality quality)
        {
            var qualityRelics = relics.Where(r => r.quality == quality).ToList();
            
            if (qualityRelics.Count == 0)
            {
                // 如果该品质没有遗物，降级选择
                return relics[Random.Range(0, relics.Count)];
            }
            
            return qualityRelics[Random.Range(0, qualityRelics.Count)];
        }
        
        /// <summary>
        /// 购买遗物
        /// </summary>
        public bool PurchaseRelic(RelicData relic)
        {
            if (Global.lotteryTicket.Value >= relic.price)
            {
                Global.lotteryTicket.Value -= relic.price;
                ownedRelics.Add(relic);
                Debug.Log($"购买遗物成功: {relic.relicName}");
                return true;
            }
            
            Debug.Log($"抽选卷不足，无法购买: {relic.relicName}");
            return false;
        }
        
        /// <summary>
        /// 触发指定类型的所有遗物效果
        /// </summary>
        public void TriggerRelics(RelicTriggerType triggerType, RelicEffectContext context)
        {
            foreach (var relic in ownedRelics)
            {
                if (relic.relicEffect != null && 
                    relic.relicEffect.IsEnabled && 
                    relic.relicEffect.CanTrigger(triggerType))
                {
                    relic.relicEffect.ExecuteEffect(context);
                }
            }
        }
        
        /// <summary>
        /// 获取已拥有的遗物列表
        /// </summary>
        public List<RelicData> GetOwnedRelics()
        {
            return new List<RelicData>(ownedRelics);
        }
    }
}
```

#### 5. 遗物商店单项UI `RelicShopItem.cs`

```csharp
using UnityEngine;
using UnityEngine.UI;
using QFramework;

namespace Gambling
{
    public class RelicShopItem : MonoBehaviour
    {
        [Header("UI元素")]
        public Image relicIcon;
        public Text relicNameText;
        public Text effectDescriptionText;
        public Text priceText;
        public Button purchaseButton;
        public Image qualityBackground;
        
        [Header("品质颜色配置")]
        public Color commonColor = Color.white;
        public Color rareColor = Color.blue;
        public Color epicColor = new Color(0.6f, 0f, 1f); // 紫色
        public Color legendaryColor = Color.yellow;
        
        private RelicData currentRelic;
        private RelicManager relicManager;
        
        public void SetRelicData(RelicData relic, RelicManager manager)
        {
            currentRelic = relic;
            relicManager = manager;
            
            // 设置UI显示
            relicIcon.sprite = relic.relicIcon;
            relicNameText.text = relic.relicName;
            effectDescriptionText.text = relic.effectDescription;
            priceText.text = $"{relic.price}";
            
            // 设置品质颜色
            qualityBackground.color = GetQualityColor(relic.quality);
            
            // 绑定购买按钮
            purchaseButton.onClick.RemoveAllListeners();
            purchaseButton.onClick.AddListener(OnPurchaseClicked);
            
            // 更新按钮状态
            UpdateButtonState();
        }
        
        private void OnPurchaseClicked()
        {
            if (relicManager.PurchaseRelic(currentRelic))
            {
                // 购买成功，禁用按钮
                purchaseButton.interactable = false;
                purchaseButton.GetComponentInChildren<Text>().text = "已购买";
            }
        }
        
        private void UpdateButtonState()
        {
            bool canPurchase = Global.lotteryTicket.Value >= currentRelic.price;
            purchaseButton.interactable = canPurchase;
        }
        
        private Color GetQualityColor(RelicQuality quality)
        {
            switch (quality)
            {
                case RelicQuality.Common: return commonColor;
                case RelicQuality.Rare: return rareColor;
                case RelicQuality.Epic: return epicColor;
                case RelicQuality.Legendary: return legendaryColor;
                default: return Color.white;
            }
        }
    }
}
```

```csharp
using UnityEngine;

namespace Gambling
{
    // 示例1：增加分数的遗物
    [CreateAssetMenu(fileName = "ScoreBoostRelic", menuName = "Gambling/RelicEffects/ScoreBoost")]
    public class ScoreBoostRelicEffect : RelicEffect
    {
        [Header("分数加成配置")]
        [SerializeField] private float scoreMultiplier = 1.2f; // 分数倍率
        
        public override string EffectName => "分数加成";
        
        public override bool CanTrigger(RelicTriggerType triggerType)
        {
            return triggerType == RelicTriggerType.OnScoreGain;
        }
        
        public override void ExecuteEffect(RelicEffectContext context)
        {
            // 增加分数
            int bonusScore = Mathf.RoundToInt(context.currentScore * (scoreMultiplier - 1f));
            context.gamblingGround.Score.Value += bonusScore;
            Debug.Log($"遗物效果触发：额外获得 {bonusScore} 分数");
        }
    }
    
    // 示例2：关卡开始时获得额外抽选卷
    [CreateAssetMenu(fileName = "ExtraTicketRelic", menuName = "Gambling/RelicEffects/ExtraTicket")]
    public class ExtraTicketRelicEffect : RelicEffect
    {
        [Header("抽选卷配置")]
        [SerializeField] private int extraTickets = 2;
        
        public override string EffectName => "额外抽选卷";
        
        public override bool CanTrigger(RelicTriggerType triggerType)
        {
            return triggerType == RelicTriggerType.OnLevelStart;
        }
        
        public override void ExecuteEffect(RelicEffectContext context)
        {
            Global.lotteryTicket.Value += extraTickets;
            Debug.Log($"遗物效果触发：获得 {extraTickets} 张额外抽选卷");
        }
    }
```

### 六、设计要点总结

1. **数据驱动**：使用`ScriptableObject`存储遗物配置，方便策划配置和调整
2. **效果分离**：遗物数据和遗物效果分离，支持灵活组合
3. **品质权重系统**：根据关卡动态调整品质概率
4. **触发机制**：使用枚举定义触发时机，支持多种触发条件
5. **UI解耦**：商店UI与管理器分离，便于维护
6. **QFramework集成**：使用`BindableProperty`、`UIPanel`等框架特性
