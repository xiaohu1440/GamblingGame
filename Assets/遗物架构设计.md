### 二、遗物系统实现

#### 2. 遗物系统实现类 `RelicSystem.cs`

```csharp
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using QFramework;

namespace Gambling
{
    public class RelicSystem : AbstractSystem, IRelicSystem
    {
        // 遗物库（所有可用的遗物）
        private List<RelicData> allRelics = new List<RelicData>();
        
        // 玩家已拥有的遗物
        private List<RelicData> ownedRelics = new List<RelicData>();
        
        protected override void OnInit()
        {
            // 从Resources文件夹加载所有遗物配置
            LoadRelicsFromResources();
        }
        
        /// <summary>
        /// 从Resources加载遗物配置
        /// </summary>
        private void LoadRelicsFromResources()
        {
            // 假设遗物配置放在 Resources/Relics 文件夹下
            RelicData[] relics = Resources.LoadAll<RelicData>("Relics");
            allRelics.AddRange(relics);
            Debug.Log($"遗物系统初始化完成，加载了 {allRelics.Count} 个遗物");
        }
        
        /// <summary>
        /// 根据当前关卡抽取遗物
        /// </summary>
        public List<RelicData> DrawRelicsForShop(int currentLevel, int count = 4)
        {
            // 获取已解锁的遗物
            var unlockedRelics = allRelics.Where(r => r.isUnlocked && 
                                                      r.unlockRequiredLevel <= currentLevel).ToList();
            
            if (unlockedRelics.Count == 0)
            {
                Debug.LogWarning("没有可用的遗物！");
                return new List<RelicData>();
            }
            
            // 计算各品质的权重
            Dictionary<RelicQuality, float> qualityWeights = CalculateQualityWeights(currentLevel);
            
            List<RelicData> drawnRelics = new List<RelicData>();
            
            for (int i = 0; i < count; i++)
            {
                RelicQuality selectedQuality = SelectQualityByWeight(qualityWeights);
                RelicData selectedRelic = SelectRelicByQuality(unlockedRelics, selectedQuality);
                
                if (selectedRelic != null)
                {
                    drawnRelics.Add(selectedRelic);
                }
            }
            
            return drawnRelics;
        }
        
        /// <summary>
        /// 计算各品质的权重（关卡越高，高品质概率越大）
        /// </summary>
        private Dictionary<RelicQuality, float> CalculateQualityWeights(int level)
        {
            Dictionary<RelicQuality, float> weights = new Dictionary<RelicQuality, float>();
            
            if (level == 1)
            {
                // 第1关：100%普通品质
                weights[RelicQuality.Common] = 100f;
                weights[RelicQuality.Rare] = 0f;
                weights[RelicQuality.Epic] = 0f;
                weights[RelicQuality.Legendary] = 0f;
            }
            else
            {
                // 每过一关，高品质概率增加
                float levelBonus = (level - 1) * 5f; // 每关增加5%
                
                weights[RelicQuality.Common] = Mathf.Max(60f - levelBonus, 20f);
                weights[RelicQuality.Rare] = Mathf.Min(25f + levelBonus * 0.5f, 40f);
                weights[RelicQuality.Epic] = Mathf.Min(10f + levelBonus * 0.3f, 25f);
                weights[RelicQuality.Legendary] = Mathf.Min(5f + levelBonus * 0.2f, 15f);
            }
            
            return weights;
        }
        
        /// <summary>
        /// 根据权重选择品质
        /// </summary>
        private RelicQuality SelectQualityByWeight(Dictionary<RelicQuality, float> weights)
        {
            float totalWeight = weights.Values.Sum();
            float randomValue = Random.Range(0f, totalWeight);
            float currentWeight = 0f;
            
            foreach (var kvp in weights)
            {
                currentWeight += kvp.Value;
                if (randomValue <= currentWeight)
                {
                    return kvp.Key;
                }
            }
            
            return RelicQuality.Common;
        }
        
        /// <summary>
        /// 从指定品质中随机选择遗物
        /// </summary>
        private RelicData SelectRelicByQuality(List<RelicData> relics, RelicQuality quality)
        {
            var qualityRelics = relics.Where(r => r.quality == quality).ToList();
            
            if (qualityRelics.Count == 0)
            {
                // 如果该品质没有遗物，降级选择
                return relics[Random.Range(0, relics.Count)];
            }
            
            return qualityRelics[Random.Range(0, qualityRelics.Count)];
        }
        
        /// <summary>
        /// 购买遗物
        /// </summary>
        public bool PurchaseRelic(RelicData relic)
        {
            if (Global.lotteryTicket.Value >= relic.price)
            {
                Global.lotteryTicket.Value -= relic.price;
                ownedRelics.Add(relic);
                Debug.Log($"购买遗物成功: {relic.relicName}");
                return true;
            }
            
            Debug.Log($"抽选卷不足，无法购买: {relic.relicName}");
            return false;
        }
        
        /// <summary>
        /// 触发指定类型的所有遗物效果
        /// </summary>
        public void TriggerRelics(RelicTriggerType triggerType, RelicEffectContext context)
        {
            foreach (var relic in ownedRelics)
            {
                if (relic.relicEffect != null && 
                    relic.relicEffect.IsEnabled && 
                    relic.relicEffect.CanTrigger(triggerType))
                {
                    relic.relicEffect.ExecuteEffect(context);
                }
            }
        }
        
        /// <summary>
        /// 获取已拥有的遗物列表
        /// </summary>
        public List<RelicData> GetOwnedRelics()
        {
            return new List<RelicData>(ownedRelics);
        }
        
        /// <summary>
        /// 添加遗物到库中（用于动态添加遗物）
        /// </summary>
        public void AddRelicToLibrary(RelicData relic)
        {
            if (!allRelics.Contains(relic))
            {
                allRelics.Add(relic);
            }
        }
        
        /// <summary>
        /// 清空已拥有的遗物（用于重新开始游戏）
        /// </summary>
        public void ClearOwnedRelics()
        {
            ownedRelics.Clear();
            Debug.Log("已清空所有已拥有的遗物");
        }
    }
}
```

#### 4. 在Controller中使用（关卡通关时打开商店）

```csharp
using UnityEngine;
using QFramework;
using System.Collections.Generic;

namespace Gambling
{
    public class LevelController : MonoBehaviour, IController
    {
        public IArchitecture GetArchitecture()
        {
            return Global.Interface;
        }
        
        /// <summary>
        /// 关卡通关时调用
        /// </summary>
        public void OnLevelComplete()
        {
            // 增加关卡
            Global.level.Value++;
            
            // 通过ISystem获取遗物系统
            var relicSystem = this.GetSystem<IRelicSystem>();
            
            // 抽取遗物
            List<RelicData> shopRelics = relicSystem.DrawRelicsForShop(Global.level.Value, 4);
            
            // 打开商店UI
            UIKit.OpenPanel<UIRelicShopPanel>(new UIRelicShopPanelData
            {
                shopRelics = shopRelics
            });
        }
    }
}
```

```csharp
using UnityEngine;
using QFramework;

namespace Gambling
{
    public class GamblingGround : MonoBehaviour, IController
    {
        public IArchitecture GetArchitecture()
        {
            return Global.Interface;
        }
        
        /// <summary>
        /// 翻牌时触发遗物效果
        /// </summary>
        public void OnCardFlipped(CardItem card)
        {
            // 创建效果上下文
            RelicEffectContext context = new RelicEffectContext
            {
                gamblingGround = this,
                currentCard = card,
                currentScore = Score.Value,
                currentLevel = Global.level.Value
            };
            
            // 通过ISystem触发遗物效果
            var relicSystem = this.GetSystem<IRelicSystem>();
            relicSystem.TriggerRelics(RelicTriggerType.OnCardFlip, context);
        }
        
        /// <summary>
        /// 关卡开始时触发遗物效果
        /// </summary>
        public void OnLevelStart()
        {
            RelicEffectContext context = new RelicEffectContext
            {
                gamblingGround = this,
                currentLevel = Global.level.Value
            };
            
            var relicSystem = this.GetSystem<IRelicSystem>();
            relicSystem.TriggerRelics(RelicTriggerType.OnLevelStart, context);
        }
```

### 五、遗物数据结构（保持不变）

#### 7. 遗物数据类 `RelicData.cs`

```csharp
using UnityEngine;
using QFramework;

namespace Gambling
{
    // 遗物品质枚举
    public enum RelicQuality
    {
        Common = 0,    // 普通（白色）
        Rare = 1,      // 稀有（蓝色）
        Epic = 2,      // 史诗（紫色）
        Legendary = 3  // 传奇（金色）
    }

    // 遗物触发时机枚举
    public enum RelicTriggerType
    {
        OnLevelStart,      // 关卡开始时
        OnLevelComplete,   // 关卡完成时
        OnCardFlip,        // 翻牌时
        OnScoreGain,       // 获得分数时
        OnTicketUse,       // 使用抽选卷时
        Passive            // 被动效果（持续生效）
    }

    [CreateAssetMenu(fileName = "NewRelic", menuName = "Gambling/RelicItem")]
    public class RelicData : ScriptableObject
    {
        [Header("基础信息")]
        public int relicId;                          // 遗物ID
        public string relicName;                     // 遗物名称
        public Sprite relicIcon;                     // 遗物图标
        
        [Header("品质与价格")]
        public RelicQuality quality;                 // 遗物品质
        public int price;                            // 遗物价格
        
        [Header("描述信息")]
        [TextArea(2, 4)]
        public string effectDescription;             // 遗物效果描述
        [TextArea(2, 4)]
        public string extraAttributeDescription;     // 额外属性文本描述
        
        [Header("触发条件")]
        public RelicTriggerType triggerType;         // 遗物触发条件
        
        [Header("解锁条件")]
        public bool isUnlocked = true;               // 遗物解锁状态
        public int unlockRequiredLevel = 1;          // 解锁所需关卡
        public string unlockConditionDescription;    // 解锁条件描述
        
        [Header("效果配置")]
        public RelicEffect relicEffect;              // 关联的遗物效果脚本
    }
}
```

### 六、使用ISystem的优势

#### 相比MonoBehaviour的RelicManager，使用ISystem有以下优势：

1. **架构一致性**：完全符合QFramework的架构设计理念
2. **无需场景依赖**：不需要在场景中放置GameObject，系统自动初始化
3. **全局访问**：在任何`IController`、`ICommand`、`IQuery`中都可以通过`this.GetSystem<IRelicSystem>()`访问
4. **生命周期管理**：由架构自动管理初始化和销毁
5. **解耦性更好**：不依赖Unity的MonoBehaviour生命周期
6. **易于测试**：可以方便地进行单元测试和Mock

#### 与EasterEggManager的对比：

- `EasterEggManager`使用MonoBehaviour是因为它需要执行协程（`StartCoroutine`）
- `RelicSystem`不需要协程，所以使用`ISystem`更合适
- 如果遗物效果需要协程，可以在`RelicEffect`中使用`ActionKit`的延迟执行功能
